# Strength of Trailing Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the *Strength of Trailing* spell: for eight minutes the caster leaves a fuse of soul-fire-like flames on the blocks they step off, harmless to the caster and their allies, burning everything else, fading one block a second from the far end.

**Architecture:** The spell plugs into the existing framework (a `SpellRules` declaration, a `MarkerEffect`, a scroll, a glyph). New is a common-side fire block, `spellcraft:trail_fire`, that reuses vanilla soul fire's models and a server-side ledger, `TrailFires`, saved as a NeoForge **level** attachment, that lays, fades and owns flames. Two pure classes in `spell.rules` hold the logic (`TrailTracker`: which block to light; `Trail`: the fuse and its countdown); thin event handlers in `TrailLayer` call them each server tick. No client code and no packets: the block renders from data, and 26.3 picks the chunk layer from the texture's transparency.

**Tech Stack:** Minecraft 26.3, NeoForge 26.3.0.23-beta (ModDevGradle 2.0.147), Java 25, JUnit 5.14.4, jqwik 1.10.1, ArchUnit 1.5.1, AssertJ 3.27.7, Error Prone 2.50.0, Spotless 8.10.3 + palantir-java-format 2.100.0, PIT 1.30.0, Python 3 stdlib for the icon script, pre-commit, gitlint, markdownlint-cli2.

**Spec:** [`docs/superpowers/specs/2026-11-04-trailing-design.md`](../specs/2026-11-04-trailing-design.md)

```java
import java.util.List;
import java.util.OptionalLong;
import net.jqwik.api.Arbitraries;
import net.jqwik.api.Arbitrary;
import net.jqwik.api.Combinators;
import net.jqwik.api.ForAll;
import net.jqwik.api.Property;
import net.jqwik.api.Provide;

class TrailTrackerPropertiesTest {
    /** One observation: a position (few distinct values, so repeats are common) and whether the player was grounded. */
    record Step(int pos, boolean grounded) {}

    @Provide
    Arbitrary<List<Step>> steps() {
        Arbitrary<Step> step = Combinators.combine(Arbitraries.integers().between(0, 3), Arbitraries.of(true, false))
                .as(Step::new);
        return step.list().ofMinSize(1).ofMaxSize(30);
    }

    @Property
    void lightsAtMostThePreviousPositionAndOnlyIfItWasGrounded(@ForAll("steps") List<Step> steps) {
        TrailTracker tracker = new TrailTracker();
        Step previous = null;
        for (Step step : steps) {
            OptionalLong lit = tracker.observe(step.pos(), step.grounded());
            if (previous == null || !previous.grounded() || previous.pos() == step.pos()) {
                assertThat(lit).as("steps %s at %s", steps, step).isEmpty();
            } else {
                assertThat(lit).isEqualTo(OptionalLong.of(previous.pos()));
            }
            previous = step;
        }
    }

    @Property
    void neverLightsTheCurrentPosition(@ForAll("steps") List<Step> steps) {
        TrailTracker tracker = new TrailTracker();
        for (Step step : steps) {
            OptionalLong lit = tracker.observe(step.pos(), step.grounded());
            assertThat(lit.isPresent() && lit.getAsLong() == step.pos()).isFalse();
        }
    }
```

```java
package io.github.willrieske.spellcraft.spell.rules;

import java.util.OptionalLong;

/**
 * Decides which block a trailing player just left. One per player. Positions are packed longs (the plumbing uses
 * {@code BlockPos.asLong()}); "grounded" is supplied by the plumbing as {@code onGround() && !isPassenger()}.
 */
public final class TrailTracker {
    private boolean observed;
    private long lastPos;
    private boolean lastGrounded;

    /**
     * Records where the player is now and returns the block to light: the previous position, when there was one,
     * the player was grounded there, and the position changed. Otherwise empty.
     *
     * @param pos the player's current block position, packed
     * @param grounded whether the player is on the ground and not riding
     * @return the previous position to light, or empty
     */
    public OptionalLong observe(long pos, boolean grounded) {
        OptionalLong toLight =
                observed && lastGrounded && pos != lastPos ? OptionalLong.of(lastPos) : OptionalLong.empty();
        observed = true;
        lastPos = pos;
        lastGrounded = grounded;
        return toLight;
    }

    /** Forgets the last observation (the effect started, or the player changed level). */
    public void reset() {
        observed = false;
    }
}
```

```java
package io.github.willrieske.spellcraft.spell.rules;

import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Collection;
import java.util.List;
import java.util.Objects;
import java.util.Optional;

/**
 * One player's trail as a fuse: flames oldest first, and a countdown until the oldest fades. Once the countdown
 * runs out the oldest flame is removed and the countdown restarts for the next one, so a trail shrinks from its far
 * end at one flame per {@link #FADE_TICKS} (or {@link #LINGER_TICKS} for a lingering flame).
 */
public final class Trail {
    /** Ticks a plain flame burns once it is the oldest. */
    public static final int FADE_TICKS = 20;
    /** Ticks a lingering flame (over soul sand or soul soil) burns once it is the oldest. */
    public static final int LINGER_TICKS = 40;

    /**
     * One flame.
     *
     * @param pos the block position, packed as by {@code BlockPos.asLong()}
     * @param lingers whether the flame takes {@link #LINGER_TICKS} rather than {@link #FADE_TICKS} to fade
     */
    public record Flame(long pos, boolean lingers) {}

    private final ArrayDeque<Flame> flames = new ArrayDeque<>();
    private int countdown;

    /** An empty trail. */
    public Trail() {}

    /**
     * Rebuilds a saved trail. The countdown is clamped into {@code 1 .. fadeTicks(oldest)} so corrupt data can never
     * stall or skip a fade; an empty trail has countdown 0.
     *
     * @param flames the flames, oldest first
     * @param countdown the saved countdown for the oldest flame
     */
    public Trail(Collection<Flame> flames, int countdown) {
        Objects.requireNonNull(flames, "flames").forEach(this::add);
        if (!this.flames.isEmpty()) {
            this.countdown = Math.clamp(countdown, 1, fadeTicks(this.flames.peekFirst()));
        }
    }

    /** How many ticks {@code flame} burns once it is the oldest. */
    public static int fadeTicks(Flame flame) {
        return flame.lingers() ? LINGER_TICKS : FADE_TICKS;
    }

    /** Appends a flame. The countdown starts only if the trail was empty. */
    public void add(Flame flame) {
        Objects.requireNonNull(flame, "flame");
        if (flames.isEmpty()) {
            countdown = fadeTicks(flame);
        }
        flames.addLast(flame);
    }

    /**
     * Advances one tick.
     *
     * @return the oldest flame if it faded this tick, otherwise empty
     */
    public Optional<Flame> tick() {
        if (flames.isEmpty()) {
            return Optional.empty();
        }
        countdown--;
        if (countdown > 0) {
            return Optional.empty();
        }
        Flame faded = flames.removeFirst();
        restartCountdown();
        return Optional.of(faded);
    }

    /**
     * Removes the oldest flames until no more than {@code max} remain. If the oldest flame changed, the countdown
     * restarts for the new oldest.
     *
     * @param max the most flames to keep (≥ 0)
     * @return the removed flames, oldest first
     */
    public List<Flame> trim(int max) {
        if (max < 0) {
            throw new IllegalArgumentException("max must be >= 0: " + max);
        }
        List<Flame> removed = new ArrayList<>();
        while (flames.size() > max) {
            removed.add(flames.removeFirst());
        }
        if (!removed.isEmpty()) {
            restartCountdown();
        }
        return removed;
    }

    private void restartCountdown() {
        countdown = flames.isEmpty() ? 0 : fadeTicks(flames.peekFirst());
    }

    /** The flames, oldest first (a copy). */
    public List<Flame> flames() {
        return List.copyOf(flames);
    }

    /** Ticks left until the oldest flame fades; 0 when empty. */
    public int countdown() {
        return countdown;
    }

    public int size() {
        return flames.size();
    }

    public boolean isEmpty() {
        return flames.isEmpty();
    }
}
```

```java
    @Override
```

```java
package io.github.willrieske.spellcraft.trailing;

import com.mojang.serialization.Codec;
import com.mojang.serialization.MapCodec;
import com.mojang.serialization.codecs.RecordCodecBuilder;
import io.github.willrieske.spellcraft.config.ServerConfig;
import io.github.willrieske.spellcraft.spell.rules.Trail;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.Iterator;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.UUID;
import net.minecraft.core.BlockPos;
import net.minecraft.core.Direction;
import net.minecraft.core.UUIDUtil;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.tags.BlockTags;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.state.BlockState;

/**
 * Every trail in one level: each owner's {@link Trail} and which flame belongs to whom. The value of the
 * {@code trailing_fires} level attachment; level attachments are re-saved on every world save (docs/api-notes.md
 * §18), so this object is mutated in place.
 */
public final class TrailFires {
    /** One owner's saved trail. */
    public record Entry(UUID owner, List<Trail.Flame> flames, int countdown) {
        static final Codec<Trail.Flame> FLAME_CODEC = RecordCodecBuilder.create(instance -> instance.group(
                        Codec.LONG.fieldOf("pos").forGetter(Trail.Flame::pos),
                        Codec.BOOL.fieldOf("lingers").forGetter(Trail.Flame::lingers))
                .apply(instance, Trail.Flame::new));

        static final Codec<Entry> CODEC = RecordCodecBuilder.create(instance -> instance.group(
                        UUIDUtil.CODEC.fieldOf("owner").forGetter(Entry::owner),
                        FLAME_CODEC.listOf().fieldOf("flames").forGetter(Entry::flames),
                        Codec.INT.fieldOf("countdown").forGetter(Entry::countdown))
                .apply(instance, Entry::new));

        public Entry {
            flames = List.copyOf(flames);
        }
    }

    public static final MapCodec<TrailFires> CODEC =
            Entry.CODEC.listOf().fieldOf("trails").xmap(TrailFires::new, TrailFires::entries);

    private final Map<UUID, Trail> byOwner = new LinkedHashMap<>();
    private final Map<Long, UUID> ownerByPos = new HashMap<>();

    /** No trails. */
    public TrailFires() {}

    /** Rebuilds the ledger from saved entries; empty trails are dropped. */
    public TrailFires(List<Entry> entries) {
        for (Entry entry : entries) {
            Trail trail = new Trail(entry.flames(), entry.countdown());
            if (trail.isEmpty()) {
                continue;
            }
            byOwner.put(entry.owner(), trail);
            for (Trail.Flame flame : trail.flames()) {
                ownerByPos.put(flame.pos(), entry.owner());
            }
        }
    }

    /** The level's ledger (created empty on first use). */
    public static TrailFires of(ServerLevel level) {
        return level.getData(TrailingAttachments.FIRES);
    }

    /** Every owner's trail, in the order the owners first lit a flame. */
    public List<Entry> entries() {
        List<Entry> entries = new ArrayList<>();
        byOwner.forEach((owner, trail) -> entries.add(new Entry(owner, trail.flames(), trail.countdown())));
        return entries;
    }

    /** Who owns the flame at {@code pos}, if the ledger knows one. */
    public Optional<UUID> ownerOf(BlockPos pos) {
        return Optional.ofNullable(ownerByPos.get(pos.asLong()));
    }

    /**
     * Lights {@code pos} for {@code owner} if the spot allows it (spec §4): air, or a replaceable block with no fluid
     * that is not fire, over a block with a sturdy top face. Then trims the owner's trail to the configured maximum.
     *
     * @return whether a flame was lit
     */
    public boolean light(ServerLevel level, ServerPlayer owner, BlockPos pos) {
        if (!canLight(level, pos)) {
            return false;
        }
        BlockPos below = pos.below();
        boolean lingers = level.getBlockState(below).is(BlockTags.SOUL_FIRE_BASE_BLOCKS);
        level.setBlock(pos, ModBlocks.TRAIL_FIRE.get().defaultBlockState(), Block.UPDATE_ALL);
        UUID id = owner.getUUID();
        Trail trail = byOwner.computeIfAbsent(id, key -> new Trail());
        trail.add(new Trail.Flame(pos.asLong(), lingers));
        ownerByPos.put(pos.asLong(), id);
        for (Trail.Flame evicted : trail.trim(ServerConfig.trailingMaxBlocks())) {
            extinguish(level, id, evicted);
        }
        return true;
    }

    static boolean canLight(ServerLevel level, BlockPos pos) {
        BlockState state = level.getBlockState(pos);
        boolean free = state.isAir()
                || (state.canBeReplaced() && state.getFluidState().isEmpty() && !state.is(BlockTags.FIRE));
        BlockPos below = pos.below();
        return free && level.getBlockState(below).isFaceSturdy(level, below, Direction.UP);
    }

    /** One fade tick for every trail in the level; trails that empty are forgotten. */
    public void fadeTick(ServerLevel level) {
        Iterator<Map.Entry<UUID, Trail>> trails = byOwner.entrySet().iterator();
        while (trails.hasNext()) {
            Map.Entry<UUID, Trail> entry = trails.next();
            entry.getValue().tick().ifPresent(flame -> extinguish(level, entry.getKey(), flame));
            if (entry.getValue().isEmpty()) {
                trails.remove();
            }
        }
    }

    /**
     * Puts out {@code owner}'s flame. If the spot has since been relit by someone else (the first flame was put out
     * by hand or water, and another owner walked there), that newer flame is theirs and stays.
     */
    private void extinguish(ServerLevel level, UUID owner, Trail.Flame flame) {
        if (!owner.equals(ownerByPos.get(flame.pos()))) {
            return;
        }
        ownerByPos.remove(flame.pos());
        BlockPos pos = BlockPos.of(flame.pos());
        if (level.getBlockState(pos).is(ModBlocks.TRAIL_FIRE.get())) {
            level.removeBlock(pos, false);
        }
    }

    /**
     * Whether the flame at {@code pos} spares {@code entity}: the owner, or (while the owner is online in this
     * level) anyone the owner {@code isAlliedTo} (same team, tamed pets). An orphan spares nobody.
     */
    public boolean isSpared(ServerLevel level, BlockPos pos, Entity entity) {
        UUID owner = ownerByPos.get(pos.asLong());
        if (owner == null) {
            return false;
        }
        if (owner.equals(entity.getUUID())) {
            return true;
        }
        Player ownerPlayer = level.getPlayerByUUID(owner);
        return ownerPlayer != null && ownerPlayer.isAlliedTo(entity);
    }
}
```

```java
package io.github.willrieske.spellcraft.trailing;

import io.github.willrieske.spellcraft.Spellcraft;
import io.github.willrieske.spellcraft.effect.ModEffects;
import io.github.willrieske.spellcraft.spell.rules.TrailTracker;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;
import net.minecraft.core.BlockPos;
import net.minecraft.resources.ResourceKey;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.level.Level;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.entity.player.PlayerEvent;
import net.neoforged.neoforge.event.tick.LevelTickEvent;
import net.neoforged.neoforge.event.tick.PlayerTickEvent;

/**
 * Lays each trailing player's flames (every server player tick) and fades every trail (every server level tick).
 * Trackers are transient: a relog or level change simply starts a fresh observation.
 */
@EventBusSubscriber(modid = Spellcraft.MODID)
public final class TrailLayer {
    /** A player's tracker and the level it observes; a level change resets it. */
    private record Tracked(ResourceKey<Level> dimension, TrailTracker tracker) {}

    private static final Map<UUID, Tracked> TRACKERS = new HashMap<>();

    private TrailLayer() {}

    @SubscribeEvent
    public static void onPlayerTick(PlayerTickEvent.Post event) {
        if (event.getEntity() instanceof ServerPlayer player) {
            track(player);
        }
    }

    @SubscribeEvent
    public static void onLevelTick(LevelTickEvent.Post event) {
        if (event.getLevel() instanceof ServerLevel level) {
            TrailFires.of(level).fadeTick(level);
        }
    }

    @SubscribeEvent
    public static void onLogout(PlayerEvent.PlayerLoggedOutEvent event) {
        TRACKERS.remove(event.getEntity().getUUID());
    }

    /**
     * One trail-laying step: observe where the player stands and light the block they just left. Without the effect
     * the player's tracker is dropped. Public so GameTests can run a step without waiting for a tick.
     */
    public static void track(ServerPlayer player) {
        UUID id = player.getUUID();
        if (!player.hasEffect(ModEffects.TRAILING)) {
            TRACKERS.remove(id);
            return;
        }
        ServerLevel level = player.level();
        Tracked tracked = TRACKERS.get(id);
        if (tracked == null || !tracked.dimension().equals(level.dimension())) {
            tracked = new Tracked(level.dimension(), new TrailTracker());
            TRACKERS.put(id, tracked);
        }
        boolean grounded = player.onGround() && !player.isPassenger();
        tracked.tracker()
                .observe(player.blockPosition().asLong(), grounded)
                .ifPresent(pos -> TrailFires.of(level).light(level, player, BlockPos.of(pos)));
    }
}
```
